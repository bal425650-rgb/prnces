const screens =

/* =========================================
   SCREEN NAVIGATION
========================================= */

function showScreen(number) {

    screens.forEach(screen => {

        screen.classList.remove("active");

    });

    const selectedScreen =
        document.getElementById(`screen${number}`);

    if (selectedScreen) {

        selectedScreen.classList.add("active");

    }

}


/* =========================================
   MUSIC
========================================= */

let musicStarted = false;


function startMusic() {

    if (musicStarted) {
        return;
    }

    backgroundMusic.volume = 0.35;

    backgroundMusic
        .play()
        .then(() => {

            musicStarted = true;

            musicToggle.textContent = "♫";

        })
        .catch(error => {

            console.log(
                "Music needs user interaction:",
                error
            );

        });

}


/* Music ON / OFF */

musicToggle.addEventListener(
    "click",
    () => {

        if (backgroundMusic.paused) {

            backgroundMusic
                .play()
                .then(() => {

                    musicStarted = true;

                    musicToggle.textContent = "♫";

                });

        } else {

            backgroundMusic.pause();

            musicToggle.textContent = "🔇";

        }

    }
);


/* =========================================
   TYPEWRITER
========================================= */

let typingTimer = null;


function typeText(element, text, speed = 25) {

    clearInterval(typingTimer);

    element.innerHTML = "";

    let index = 0;

    const caret =
        document.createElement("span");

    caret.className = "caret";

    element.appendChild(caret);


    typingTimer =
        setInterval(() => {

            if (index >= text.length) {

                clearInterval(typingTimer);

                typingTimer = null;

                caret.remove();

                return;
            }


            const character =
                text.charAt(index);


            if (character === "\n") {

                caret.before(
                    document.createElement("br")
                );

            } else {

                caret.before(
                    document.createTextNode(character)
                );

            }


            index++;

        }, speed);

}


/* =========================================
   FALLING PETALS
========================================= */

let petalInterval = null;


function createPetal() {

    const petal =
        document.createElement("span");

    petal.className =
        "falling-petal";


    /* Random horizontal position */

    petal.style.left =
        Math.random() * 100 + "vw";


    /* Random size */

    const size =
        Math.random() * 10 + 8;

    petal.style.width =
        size + "px";

    petal.style.height =
        size * 1.4 + "px";


    /* Random falling speed */

    const duration =
        Math.random() * 6 + 7;


    petal.style.animationDuration =
        `${duration}s, ${Math.random() * 2 + 2}s`;


    petal.style.animationDelay =
        Math.random() * 2 + "s";


    petalField.appendChild(petal);


    setTimeout(() => {

        petal.remove();

    }, (duration + 3) * 1000);

}


function startPetals() {

    if (petalInterval) {
        return;
    }

    petalInterval =
        setInterval(createPetal, 700);

}


function stopPetals() {

    if (petalInterval) {

        clearInterval(petalInterval);

        petalInterval = null;

    }

}
